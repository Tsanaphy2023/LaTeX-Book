\begin{lessonplanbox}
\noindent\textbf{หัวข้อเนื้อหาประจำบท}
\begin{enumerate}
    \item อันตรกิริยามูลฐาน 4 แรงในธรรมชาติ
    \item แบบจำลองมาตรฐาน
    \item ฮาดรอนและโครงสร้างควาร์ก
    \item แผนภาพไฟน์แมน
    \item สถาปัตยกรรมการจำลองเสมือนจริง AR/XR 3D Hadron Structure
    \item การคำนวณเชิงตัวเลขด้วย Python Invariant Mass Reconstruction
    \item ตัวอย่างการแก้โจทย์เชิงวิเคราะห์ตามระเบียบวิธี P-S-C-T
    \item สรุปสาระสำคัญและแบบฝึกหัดท้ายบท
\end{enumerate}

\noindent\textbf{วัตถุประสงค์เชิงพฤติกรรม}
\begin{enumerate}
    \item จำแนกอันตรกิริยามูลฐานทั้ง 4 แรงในธรรมชาติและอนุภาคสื่อแรง (Gauge Bosons) ได้อย่างถูกต้อง
    \item อธิบายองค์ประกอบของแบบจำลองมาตรฐาน ทั้งแฟร์มิออน (ควาร์กและเลปตอน) และโบซอนได้อย่างเป็นระบบ
    \item วิเคราะห์กฎการอนุรักษ์ในปฏิกิริยาอนุภาค ได้แก่ ประจุไฟฟ้า เลขแบริออน เลขเลปตอน และสเตรนจ์เนสได้อย่างแม่นยำ
    \item วาดและตีความแผนภาพไฟน์แมนสำหรับอันตรกิริยาแม่เหล็กไฟฟ้าและแรงนิวเคลียร์อย่างอ่อนได้อย่างถูกต้อง
    \item อธิบายบทบาทของสนามฮิกส์ (Higgs Field) และกลไกการสร้างมวลแก่อนุภาคมูลฐานได้อย่างถูกต้อง
    \item พัฒนาโปรแกรมภาษา Python เพื่อคำนวณการอนุรักษ์ 4-โมเมนตัม และการสร้างมวลคงตัว (Invariant Mass Reconstruction) ได้อย่างมีประสิทธิภาพ
\end{enumerate}

\noindent\textbf{กิจกรรมการเรียนการสอน}
\begin{enumerate}
    \item การบรรยายเชิงทฤษฎีประกอบสื่อมัลติมีเดียเรื่องแบบจำลองมาตรฐานและอนุภาคมูลฐาน
    \item กิจกรรมกลุ่มย่อยวิเคราะห์กฎการอนุรักษ์ในปฏิกิริยาการสลายตัวของอนุภาคและการวาดแผนภาพไฟน์แมน
    \item ปฏิบัติการจำลองโครงสร้างควาร์ก 3 มิติ (3D Hadron Structure) บนแพลตฟอร์ม WebXR
    \item ปฏิบัติการคอมพิวเตอร์เขียนโค้ด Python เพื่อวิเคราะห์ข้อมูลการชนและการสร้างมวลคงตัว
    \item การศึกษาและแก้โจทย์ปัญหาฟิสิกส์อนุภาคตามระเบียบวิธี P-S-C-T
\end{enumerate}

\noindent\textbf{สื่อการเรียนการสอน}
\begin{enumerate}
    \item เอกสารคำสอน บทที่ 3 ฟิสิกส์อนุภาคและแบบจำลองมาตรฐาน
    \item สื่อจำลอง 3 มิติปฏิสัมพันธ์ WebXR 3D Hadron Structure Simulator
    \item ชุดสคริปต์ภาษา Python คำนวณมวลคงตัวและจำลองการกระจายของฮิกส์โบซอน
    \item แพลตฟอร์มการจัดการเรียนรู้ดิจิทัล (LMS) และชุดโจทย์แบบฝึกหัด P-S-C-T

\noindent\textbf{การประเมินผล}

    \item การตรวจรายงานผลปฏิบัติการคอมพิวเตอร์ด้วย Python และการวาดแผนภาพไฟน์แมน

\section{แบบจำลองมาตรฐาน}
\index{ควาร์ก}
\index{เลปตอน}
\index{แฟร์มิออน}

\textbf{แบบจำลองมาตรฐาน (The Standard Model)} คือทฤษฎีควอนตัมสนาม (Quantum Field Theory) ที่อธิบายอนุภาคพื้นฐานที่เป็นองค์ประกอบของสสารและแรงสื่อสาร (ยกเว้นแรงโน้มถ่วง) อนุภาคในแบบจำลองมาตรฐานแบ่งเป็น 2 กลุ่มใหญ่ตามค่าสปิน ดังนี้

\subsection{แฟร์มิออน}
\textbf{แฟร์มิออน สปิน $s = 1/2$} เป็นอนุภาคที่สร้างสสาร เป็นไปตามหลักการกีดกันของเพาลี (Pauli Exclusion Principle) แบ่งออกเป็น 3 ยุค (Generations) ดังนี้
\begin{itemize}
    \item \textbf{ควาร์ก (Quarks)} มีประจุเศษส่วนและมีประจุสี (แดง เขียว น้ำเงิน) ประกอบด้วย
    \begin{itemize}
        \item รุ่นที่ 1 ได้แก่ อัพ ($u, +2/3e$) และ ดาวน์ ($d, -1/3e$)
        \item รุ่นที่ 2 ได้แก่ ชาร์ม ($c, +2/3e$) และ สเตรนจ์ ($s, -1/3e$)
        \item รุ่นที่ 3 ได้แก่ ท็อป ($t, +2/3e$) และ บอตทอม ($b, -1/3e$)
    \end{itemize}
    \item \textbf{เลปตอน (Leptons)} ไม่มีประจุสี ไม่ทำอันตรกิริยาอย่างเข้ม ประกอบด้วย
    \begin{itemize}
        \item รุ่นที่ 1 ได้แก่ อิเล็กตรอน ($e^-$) และ อิเล็กตรอนนิวทริโน ($\nu_e$)
        \item รุ่นที่ 2 ได้แก่ มิวออน ($\mu^-$) และ มิวออนนิวทริโน ($\nu_\mu$)
        \item รุ่นที่ 3 ได้แก่ ทาว ($\tau^-$) และ ทาวน์นิวทริโน ($\nu_\tau$)
    \end{itemize}
\end{itemize}

\subsection{เกจโบซอน}
\textbf{เกจโบซอน สปิน $s = 1$} อนุภาคสื่อแรงตามสมมาตรเกจ $SU(3)_C \times SU(2)_L \times U(1)_Y$ ได้แก่ โฟตอน ($\gamma$), กลูออน ($g$), โบซอนเวกเตอร์ $W^{\pm}$ และ $Z^0$

\subsection{สเกลาร์โบซอน ฮิกส์โบซอน}
\textbf{ฮิกส์โบซอน สปิน $s = 0$} ค้นพบที่เซิร์น (CERN LHC) ในปี ค.ศ. 2012 มวลประมาณ $125.1\text{ GeV}/c^2$ ทำหน้าที่ให้มวลแก่อนุภาคผ่านกลไกการทำลายสมมาตรทางไฟฟ้า-อ่อนอย่างเกิดขึ้นเอง (Spontaneous Symmetry Breaking)

\section{ฮาดรอนและโครงสร้างควาร์ก}
\index{ฮาดรอน}
\index{แบริออน}
\index{เมซอน}
\index{การกักขังประจุสี}

\textbf{ฮาดรอนและโครงสร้างควาร์ก} เนื่องจากปรากฏการณ์ \textbf{การกักขังประจุสี (Color Confinement)} ในทฤษฎีควอนตัมโครโมไดนามิกส์ (QCD) ควาร์กเดี่ยวไม่สามารถปรากฏอยู่อย่างอิสระได้ ควาร์กจึงต้องรวมตัวกันเป็นกลุ่มที่ประจุสีรวมเป็นกลาง เรียกว่า \textbf{ฮาดรอน (Hadrons)} ดังนี้
\begin{enumerate}
    \item \textbf{แบริออน (Baryons)} ประกอบด้วย 3 ควาร์ก ($qqq$) เช่น
    \begin{itemize}
        \item โปรตอน ($p$) มีโครงสร้างควาร์กเป็น $uud$ (ประจุรวม $= +2/3 + 2/3 - 1/3 = +1$)
        \item นิวตรอน ($n$) มีโครงสร้างควาร์กเป็น $udd$ (ประจุรวม $= +2/3 - 1/3 - 1/3 = 0$)
    \end{itemize}
    \item \textbf{เมซอน (Mesons)} ประกอบด้วยควาร์กและแอนติควาร์ก ($q\bar{q}$) เช่น ไพออนบวก ($\pi^+ = u\bar{d}$)
\end{enumerate}

\section{การคำนวณเชิงตัวเลขด้วย Python Invariant Mass Reconstruction}
\index{Python}
\index{LHC}
\index{มวลคงตัว}

ในฟิสิกส์พลังงานสูง เมื่ออนุภาคที่ไม่เสถียร (เช่น ฮิกส์โบซอน $H^0$) สลายตัวอย่างรวดเร็วเป็นสองโฟตอน ($H^0 \to \gamma_1 + \gamma_2$) นักฟิสิกส์จะคำนวณ \textbf{มวลคงตัว (Invariant Mass, $M$)}
\begin{equation}
M^2 c^4 = (E_1 + E_2)^2 - (\mathbf{p}_1 + \mathbf{p}_2)^2 c^2 = 2 E_1 E_2 (1 - \cos\theta)
\end{equation}

\begin{pythonbox}[invariant\_mass\_higgs.py -- การวิเคราะห์สเปกตรัมมวลคงตัวของฮิกส์โบซอน]
import numpy as np

def simulate_higgs_reconstruction(num_events=100000):
    """
    จำลองการวัดมวลคงตัวจากการสลายตัวของฮิกส์โบซอนเป็น 2 โฟตอน
    M_Higgs = 125.1 GeV, การแจกแจงแบบ Breit-Wigner ร่วมกับ Gaussian Detector Resolution
    """
    np.random.seed(42)
    m_higgs_true = 125.1  # GeV/c^2
    gamma_width = 0.004   # ความกว้างธรรมชาติของฮิกส์ (GeV)
    resolution = 1.5      # ความคลาดเคลื่อนของเครื่องตรวจวัด (GeV)
    
    # สุ่มเหตุการณ์สัญญาณ (Signal Events)
    sig_mass = np.random.normal(m_higgs_true, resolution, int(num_events * 0.05))
    
    # สุ่มสัญญาณพื้นหลังแบบเอ็กซ์โพเนนเชียล (Background Events)
    bg_mass = np.random.exponential(scale=35.0, size=int(num_events * 0.95)) + 100.0
    
    # รวมข้อมูลทั้งหมด
    total_data = np.concatenate([sig_mass, bg_mass])
    total_data = total_data[(total_data >= 110.0) & (total_data <= 140.0)]
    
    counts, bin_edges = np.histogram(total_data, bins=60, range=(110, 140))
    peak_bin = np.argmax(counts)
    peak_mass = 0.5 * (bin_edges[peak_bin] + bin_edges[peak_bin+1])
    
    print(f"=== Higgs Invariant Mass Analysis ===")
    print(f"Total detected events in [110, 140] GeV: {len(total_data)}")
    print(f"Reconstructed Peak Mass: {peak_mass:.2f} GeV/c^2 (Expected: 125.10 GeV/c^2)")

simulate_higgs_reconstruction()
\end{pythonbox}

\section{ตัวอย่างการแก้โจทย์เชิงวิเคราะห์ตามระเบียบวิธี P-S-C-T}

\begin{psctexample}[พลังงานปลดปล่อยในการสลายตัวของไพออนนิ่ง]
\textbf{โจทย์ (Problem)} \\
ไพออนบวก ($\pi^+$) ขณะอยู่นิ่ง เกิดการสลายตัวผ่านอันตรกิริยาอย่างอ่อนเป็นมิวออนบวกและมิวออนนิวทริโน
\begin{equation}
\pi^+ \to \mu^+ + \nu_\mu
\end{equation}
กำหนด\mbox{มวลนิ่ง} $m_\pi = 139.57\text{ MeV}/c^2$, $m_\mu = 105.66\text{ MeV}/c^2$ และสมมติให้นิวทริโนไร้มวล ($m_\nu \approx 0$)
\begin{enumerate}
    \item จงคำนวณโมเมนตัมของมิวออนและนิวทริโนที่เกิดขึ้น
    \item จงหาพลังงานจลน์ของมิวออน
\end{enumerate}

\textbf{ยุทธวิธี (Strategy)} \\
เนื่องจากไพออนตั้งต้นอยู่นิ่ง โมเมนตัมรวมตั้งต้นเป็นศูนย์
\begin{equation}
\mathbf{p}_\mu + \mathbf{p}_\nu = 0 \implies |\mathbf{p}_\mu| = |\mathbf{p}_\nu| = p
\end{equation}
การอนุรักษ์พลังงานรวมสัมพัทธภาพ
\begin{equation}
E_\pi = m_\pi c^2 = E_\mu + E_\nu = \sqrt{(pc)^2 + (m_\mu c^2)^2} + pc
\end{equation}

\textbf{การคำนวณ (Computation)} \\
จัดรูปสมการเพื่อหา $pc$
\begin{align}
m_\pi c^2 - pc &= \sqrt{(pc)^2 + (m_\mu c^2)^2} \\
(m_\pi c^2)^2 - 2 (m_\pi c^2)(pc) + (pc)^2 &= (pc)^2 + (m_\mu c^2)^2 \\
pc &= \frac{(m_\pi c^2)^2 - (m_\mu c^2)^2}{2 m_\pi c^2}
\end{align}
แทนค่าตัวเลขในหน่วย MeV
\begin{align}
pc &= \frac{(139.57)^2 - (105.66)^2}{2 (139.57)} = \frac{19479.78 - 11164.04}{279.14} \nonumber \\
&= \frac{8315.74}{279.14} \approx 29.79\text{ MeV}
\end{align}
ดังนั้น โมเมนตัมของทั้งสองอนุภาคคือ $p \approx 29.79\text{ MeV}/c$

พลังงานรวมของมิวออนคือ
\begin{align}
E_\mu &= \sqrt{(pc)^2 + (m_\mu c^2)^2} = \sqrt{(29.79)^2 + (105.66)^2} \nonumber \\
&\approx \sqrt{887.44 + 11164.04} \approx 109.78\text{ MeV}
\end{align}
พลังงานจลน์ของมิวออนคำนวณจาก
\begin{equation}
K_\mu = E_\mu - m_\mu c^2 = 109.78 - 105.66 = 4.12\text{ MeV}
\end{equation}

\textbf{ข้อคิดทางฟิสิกส์ (Takeaway)} \\
พลังงานปลดปล่อยทั้งหมดของปฏิกิริยานี้คือ $Q = (m_\pi - m_\mu)c^2 \approx 33.91\text{ MeV}$ ทว่ามิวออนได้รับพลังงานจลน์ไปเพียง $4.12\text{ MeV}$ เท่านั้น พลังงานส่วนใหญ่ถึง $29.79\text{ MeV}$ (คิดเป็น 88\%) ถูกถ่ายทอดไปให้นิวทริโนที่ไร้มวล นี่คือลักษณะเฉพาะของการอนุรักษ์โมเมนตัมในระบบอนุภาคสัมพัทธภาพ
\qedexam
\end{psctexample}

\begin{psctexample}[การสลายตัวของอนุภาคฮิกส์โบซอนและมวลไม่แปรเปลี่ยน]
ที่เครื่องชนอนุภาคขนาดใหญ่ LHC ที่เซิร์น (CERN) อนุภาคฮิกส์โบซอน $H^0$ ที่ถูกสร้างขึ้นจากการชนของโปรตอน สามารถสลายตัวไปเป็นโฟตอนพลังงานสูง 2 อนุภาคตามปฏิกิริยา
\begin{equation}
H^0 \to \gamma_1 + \gamma_2
\end{equation}
เครื่องตรวจวัดอนุภาคคาโลริมิเตอร์ (Calorimeter) ตรวจวัดพลังงานของโฟตอนทั้งสองได้ $E_1 = 70.0\text{ GeV}$ และ $E_2 = 56.5\text{ GeV}$ โดยแนวการเคลื่อนที่ของโฟตอนทั้งสองทำมุมเปิดต่อกัน $\theta = 150.0^\circ$
\begin{enumerate}
    \item จงพิสูจน์ความสัมพันธ์ของมวลไม่แปรเปลี่ยน (Invariant Mass) ของระบบโฟตอนคู่
    \begin{equation}
    M_{\gamma\gamma} c^2 = \sqrt{2 E_1 E_2 (1 - \cos\theta)}
    \end{equation}
    \item คำนวณมวลไม่แปรเปลี่ยน $M_{\gamma\gamma} c^2$ ในหน่วย $\text{GeV}$ และเปรียบเทียบกับมวลของอนุภาคฮิกส์ที่วัดได้ในระดับสากล ($125.1\text{ GeV}/c^2$)
\end{enumerate}

\textbf{โจทย์ (Problem)} \\
คำนวณ\mbox{มวลนิ่ง}ของอนุภาคแม่ที่สลายตัวจากพลังงานและมุมเปิดของโฟตอนลูกสองอนุภาค

\textbf{ยุทธวิธี (Strategy)} \\
\begin{itemize}
    \item โฟตอนเป็นอนุภาคไร้มวล มี $m_\gamma = 0$ ดังนั้น $E_i = p_i c$ เวกเตอร์ 4-โมเมนตัมของโฟตอนแต่ละตัวคือ $P_i = (E_i/c, \mathbf{p}_i)$ โดยที่ $P_i^2 = (E_i/c)^2 - |\mathbf{p}_i|^2 = 0$
    \item 4-โมเมนตัมรวมของระบบโฟตอนคู่คือ $P_{\text{tot}} = P_1 + P_2$
    \item มวลไม่แปรเปลี่ยนคำนวณจากสเกลาร์ลอเรนซ์ $P_{\text{tot}}^2 = M_{\gamma\gamma}^2 c^2$
    \begin{equation}
    P_{\text{tot}}^2 = (P_1 + P_2)^2 = P_1^2 + P_2^2 + 2 P_1 \cdot P_2 = 0 + 0 + 2 \left(\frac{E_1 E_2}{c^2} - \mathbf{p}_1 \cdot \mathbf{p}_2\right)
    \end{equation}
    \item เนื่องจาก $\mathbf{p}_1 \cdot \mathbf{p}_2 = |\mathbf{p}_1| |\mathbf{p}_2| \cos\theta = \left(\frac{E_1}{c}\right)\left(\frac{E_2}{c}\right)\cos\theta$
\end{itemize}

\textbf{การคำนวณ (Computation)} \\
\textit{1. พิสูจน์สูตรมวลไม่แปรเปลี่ยน}
\begin{align}
M_{\gamma\gamma}^2 c^2 &= 2 \left[\frac{E_1 E_2}{c^2} - \frac{E_1 E_2}{c^2}\cos\theta\right] = \frac{2 E_1 E_2}{c^2}(1 - \cos\theta) \nonumber \\
M_{\gamma\gamma} c^2 &= \sqrt{2 E_1 E_2 (1 - \cos\theta)}
\end{align}

\textit{2. แทนค่าตัวเลข}
\begin{align}
\cos(150.0^\circ) &= -\frac{\sqrt{3}}{2} \approx -0.8660 \nonumber \\
1 - \cos(150.0^\circ) &= 1 - (-0.8660) = 1.8660
\end{align}
คำนวณมวล
\begin{align}
M_{\gamma\gamma} c^2 &= \sqrt{2 \times (70.0\text{ GeV}) \times (56.5\text{ GeV}) \times 1.8660} \nonumber \\
&= \sqrt{2 \times 3955 \times 1.8660} = \sqrt{14760.06} \approx 121.5\text{ GeV}
\end{align}
ซึ่งสอดคล้องกับพีคมวลของอนุภาคฮิกส์โบซอน (Higgs Boson Resonance Peak) ที่มีค่าเฉลี่ยประมาณ $125.1\text{ GeV}/c^2$ ภายในขอบเขตความคลาดเคลื่อนเชิงสถิติของการทดลอง ATLAS และ CMS

\textbf{ข้อคิดทางฟิสิกส์ (Takeaway)} \\
อนุภาคฮิกส์โบซอนมีอายุขัยสั้นมากเพียง $1.6 \times 10^{-22}\text{ s}$ จึงไม่สามารถตรวจวัดรอยทาง (Track) ของมันได้โดยตรง แต่นักฟิสิกส์อนุภาคค้นพบมันได้จากการพล็อตกราฟการแจกแจงมวลไม่แปรเปลี่ยน $M_{\gamma\gamma}$ ของโฟตอนคู่จำนวนนับล้านคู่ แล้วมองหา "ยอดพีคส่วนเกิน (Bump/Resonance)" เหนือระดับสัญญาณพื้นหลัง (Background)
\qedexam
\end{psctexample}

\begin{psctexample}[การแปลงพิกัดและวางแท่นตรวจวัดอนุภาคด้วย WebXR Hit-Test (WebXR Spatial Reticle Transformation)]
ในเซสชัน WebXR Augmented Reality การวางแบบจำลองเครื่องตรวจวัดอนุภาคหรือแบบจำลองฮาดรอน 3 มิติลงบนโต๊ะทดลองในโลกจริง อาศัยระบบตรวจจับระนาบพื้นผิวด้วยลำรังสีเสมือน (XR Hit-Testing) ซึ่งระบบจะส่งเมทริกซ์การแปลงพิกัดขนาด $4 \times 4$ แบบ Column-Major
\begin{equation}
\mathbf{M}_{\text{hit}} = \begin{bmatrix}
r_{11} & r_{12} & r_{13} & t_x \\
r_{21} & r_{22} & r_{23} & t_y \\
r_{31} & r_{32} & r_{33} & t_z \\
0 & 0 & 0 & 1
\end{bmatrix}
\end{equation}
กำหนดให้เวกเตอร์ตำแหน่งของกล้องบนแว่น AR หรือสมาร์ตโฟนคือ $\mathbf{p}_{\text{cam}} = (0.00, 1.60, 0.00)\text{ m}$ และทิศทางการมอง (Gaze Unit Vector) ของผู้สังเกตคือ $\hat{\mathbf{v}} = (0.60, -0.30, -0.7416)$
\begin{enumerate}
    \item จงอธิบายระเบียบวิธีสลายเมทริกซ์ (Matrix Decomposition) ใน Three.js เพื่อแยกเวกเตอร์ตำแหน่ง $\mathbf{t}$, ควอเทอร์เนียนการหมุน $\mathbf{q}$, และสเกล $\mathbf{s}$
    \item หากลำรังสีการมองกระทบระนาบโต๊ะทดลองที่ระยะห่างเชิงเส้นตามแนวสายตา $d = 1.80\text{ m}$ จงคำนวณเวกเตอร์พิกัด 3 มิติ $\mathbf{p}_{\text{hit}}$ ในระบบอ้างอิงของโลกเสมือน (World Coordinates) ที่ต้องนำโมเดลเครื่องตรวจวัดไปวาง
\end{enumerate}

\textbf{โจทย์ (Problem)} \\
คำนวณตำแหน่ง 3 มิติในโลกจริงที่เส้นสายตาตัดกับระนาบโต๊ะทดลองเพื่อปักหมุดวัตถุเสมือนใน WebXR

\textbf{ยุทธวิธี (Strategy)} \\
\begin{itemize}
    \item ตำแหน่งตกกระทบของการยิงรังสี (Ray Casting) คำนวณจากสมการเส้นตรงพาราเมตริก
    \begin{equation}
    \mathbf{p}_{\text{hit}} = \mathbf{p}_{\text{cam}} + d \hat{\mathbf{v}}
    \end{equation}
    \item ตรวจสอบว่า $\hat{\mathbf{v}}$ เป็นเวกเตอร์หนึ่งหน่วย $|\hat{\mathbf{v}}|^2 = (0.60)^2 + (-0.30)^2 + (-0.7416)^2 \approx 0.36 + 0.09 + 0.55 = 1.00$
    \item ใน Three.js การอัปเดตตำแหน่งของ Mesh ทำได้โดยตรงผ่านเมธอด \texttt{mesh.position.copy(pHit)} หรือใช้ \texttt{mesh.matrix.fromArray(hitPoseMatrix)}
\end{itemize}

\textbf{การคำนวณ (Computation)} \\
\textit{1. คำนวณพิกัดแต่ละแกน}
\begin{align}
x_{\text{hit}} &= x_{\text{cam}} + d v_x = 0.00 + (1.80\text{ m}) (0.60) = +1.08\text{ m} \nonumber \\
y_{\text{hit}} &= y_{\text{cam}} + d v_y = 1.60 + (1.80\text{ m}) (-0.30) = 1.60 - 0.54 = +1.06\text{ m} \nonumber \\
z_{\text{hit}} &= z_{\text{cam}} + d v_z = 0.00 + (1.80\text{ m}) (-0.7416) \approx -1.335\text{ m}
\end{align}
ดังนั้น เวกเตอร์ตำแหน่งที่ตกกระทบคือ
\begin{equation}
\mathbf{p}_{\text{hit}} = (1.08, 1.06, -1.335)\text{ m}
\end{equation}

\textit{2. การจัดวางระนาบสัมผัส (Surface Normal Alignment)}
ความสูง $y_{\text{hit}} = 1.06\text{ m}$ สอดคล้องกับความสูงระดับโต๊ะทดลองมาตรฐาน (ประมาณ $1.0\text{ m}$ จากพื้นห้อง) เมทริกซ์การหมุนจะปรับเวกเตอร์แนวฉากของเป้าเล็ง (Reticle) ให้ขนานกับเวกเตอร์แนวฉากของพื้นผิวโต๊ะ $\hat{\mathbf{n}} = (0, 1, 0)$ โดยอัตโนมัติ

\textbf{ข้อคิดทางฟิสิกส์และวิศวกรรม (Takeaway)} \\
การคำนวณพิกัดสัมบูรณ์ในระบบพิกัดคาร์ทีเซียน 3 มิติช่วยให้นักศึกษาฟิสิกส์สามารถเชื่อมโยงแนวคิดคณิตศาสตร์เวกเตอร์และเมทริกซ์เข้ากับการทำงานจริงของฮาร์ดแวร์เซนเซอร์ LiDAR และกล้อง Stereo Tracking ในอุปกรณ์ AR ได้อย่างเป็นรูปธรรม
\qedexam
\end{psctexample}

\section{แบบฝึกหัดท้ายบท}
\begin{enumerate}
    \item จงระบุว่าปฏิกิริยาอนุภาคต่อไปนี้สามารถเกิดขึ้นได้หรือไม่ หากเกิดไม่ได้ให้ระบุกฎการอนุรักษ์ที่ถูกละเมิด
    \begin{enumerate}
        \item $p \to e^+ + \pi^0$
        \item $\pi^- + p \to K^0 + \Lambda^0$
        \item $\mu^- \to e^- + \gamma$
    \end{enumerate}
    \item จงเขียนโครงสร้างควาร์กของอนุภาคแอนติโปรตอน ($\bar{p}$) และแอนตินิวตรอน ($\bar{n}$) พร้อมตรวจสอบประจุไฟฟ้ารวม
    \item ในการสลายตัวของ $Z^0$ โบซอนที่อยู่นิ่งเป็นคู่อิเล็กตรอน-โพซิตรอน ($Z^0 \to e^- + e^+$) จงคำนวณพลังงานจลน์ของอิเล็กตรอน (กำหนด $M_Z \approx 91.19\text{ GeV}/c^2$)
    \item อนุภาคแลมบ์ดา ($\Lambda^0$) สลายตัวผ่านอันตรกิริยาอย่างอ่อนตามปฏิกิริยา $\Lambda^0 \to p + \pi^-$ จงวิเคราะห์การอนุรักษ์ประจุไฟฟ้า เลขแบริออน เลขเลปตอน สปิน และเลขสเตรนจ์เนส ($\Delta S$) พร้อมอธิบายว่าเหตุใด $\Delta S = 1$ จึงยอมรับได้ในอันตรกิริยานี้
    \item จงร่างแผนภาพไฟน์แมน (Feynman Diagram) ในระดับควาร์กสำหรับการสลายตัวบีตาลบของนิวตรอน $n \to p + e^- + \bar{\nu}_e$ โดยระบุบทบาทของเกจโบซอน $W^-$ และการเปลี่ยนเฟลเวอร์ของควาร์ก ($d \to u$) อย่างละเอียด
\end{enumerate}
