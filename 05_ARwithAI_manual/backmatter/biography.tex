\begin{minipage}[t]{0.48\textwidth}
    \textbf{\large ประวัติการศึกษา}
    \begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
        \item \textbf{ปร.ด. (ฟิสิกส์ประยุกต์)}\\
        มหาวิทยาลัยเทคโนโลยีสุรนารี
        \item \textbf{วท.ม. (ฟิสิกส์)}\\
        มหาวิทยาลัยเชียงใหม่
        \item \textbf{วท.บ. (ฟิสิกส์ เกียรตินิยม)}\\
        มหาวิทยาลัยเชียงใหม่
    \end{itemize}

    \vspace{0.4cm}
    \textbf{\large ตำแหน่งทางวิชาการและบริหาร}
    \begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
        \item ผู้ช่วยศาสตราจารย์ ประจำสาขาวิชาฟิสิกส์
        \item กรรมการบริหารหลักสูตรวิทยาศาสตรบัณฑิต
        \item ผู้เชี่ยวชาญเทคโนโลยีเสมือนจริงและปัญญาประดิษฐ์ มหาวิทยาลัยราชภัฏรำไพพรรณี
    \end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\textwidth}
    \textbf{\large ความเชี่ยวชาญและงานวิจัยที่สนใจ}
    \begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
        \item \textbf{Spatial Computing \& WebAR:}\\
        การประมวลผลเชิงพื้นที่และการพัฒนาแบบจำลองความจริงเสริมสำหรับการศึกษาวิทยาศาสตร์
        \item \textbf{Computer Vision \& AI:}\\
        การตรวจจับโครงร่างมือ (Hand Tracking), ท่าทางชีวมิติ และการจำแนกภาพด้วย Deep Learning
        \item \textbf{Computational Physics:}\\
        การคำนวณเชิงตัวเลข กลศาสตร์ควอนตัม และฟิสิกส์ของแข็ง
        \item \textbf{IoT \& Smart Sensor Systems:}\\
        การตรวจวัดคุณภาพสิ่งแวดล้อมและเกษตรอัจฉริยะ
    \end{itemize}
\end{minipage}
