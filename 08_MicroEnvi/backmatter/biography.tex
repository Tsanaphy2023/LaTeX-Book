\noindent\textbf{ผู้ช่วยศาสตราจารย์ ดร.จิรภัทร จันทมาลี}
\begin{itemize}
    \item \textbf{ตำแหน่งปัจจุบัน:} ผู้ช่วยศาสตราจารย์ สาขาวิชาจุลชีววิทยา คณะวิทยาศาสตร์และเทคโนโลยี มหาวิทยาลัยราชภัฏรำไพพรรณี
    \item \textbf{คุณวุฒิการศึกษา:} 
    \begin{itemize}
        \item ปรัชญาดุษฎีบัณฑิต (ปร.ด.) สาขาจุลชีววิทยา
        \item วิทยาศาสตรมหาบัณฑิต (วท.ม.) สาขาจุลชีววิทยา
        \item วิทยาศาสตรบัณฑิต (วท.บ.) สาขาจุลชีววิทยา
    \end{itemize}
    \item \textbf{ความเชี่ยวชาญ:} จุลชีววิทยาสิ่งแวดล้อม, เทคโนโลยีการหมัก, การบำบัดสารปนเปื้อนทางชีวภาพ (Bioremediation), และการประยุกต์ใช้จุลินทรีย์ในอุตสาหกรรมเกษตร
\end{itemize}

\noindent\textbf{ผู้ช่วยศาสตราจารย์ ดร.ชีวะ ทัศนา}
\begin{itemize}
    \item \textbf{ตำแหน่งปัจจุบัน:} ผู้ช่วยศาสตราจารย์ สาขาวิชาฟิสิกส์ คณะวิทยาศาสตร์และเทคโนโลยี มหาวิทยาลัยราชภัฏรำไพพรรณี
    \item \textbf{คุณวุฒิการศึกษา:}
    \begin{itemize}
        \item ปรัชญาดุษฎีบัณฑิต (ปร.ด.) สาขาฟิสิกส์
        \item วิทยาศาสตรมหาบัณฑิต (วท.ม.) สาขาฟิสิกส์
        \item วิทยาศาสตรบัณฑิต (วท.บ.) สาขาฟิสิกส์
    \end{itemize}
    \item \textbf{ความเชี่ยวชาญ:} ฟิสิกส์สิ่งแวดล้อม, ระบบตรวจวัดอนุภาคและคุณภาพอากาศ, ปัญญาประดิษฐ์และไอโอทีเพื่อการเกษตรแม่นยำ (AIoT), และเทคโนโลยีโลกเสมือน (AR/XR Scientific Simulations)
\end{itemize}
