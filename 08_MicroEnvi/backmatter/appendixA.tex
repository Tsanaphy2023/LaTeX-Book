\chapter{สูตรอาหารเลี้ยงเชื้อและการเตรียมสีย้อมมาตรฐาน}
\label{app:appendixA}

\section{สูตรอาหารเลี้ยงเชื้อมาตรฐาน}
\noindent\textbf{1. Nutrient Agar (NA)}
\begin{itemize}
    \item Beef extract 3.0 g
    \item Peptone 5.0 g
    \item Agar 15.0 g
    \item น้ำกลั่น ปรับปริมาตรเป็น 1,000 mL (pH $6.8 \pm 0.2$)
\end{itemize}

\noindent\textbf{2. MacConkey Agar (MAC)}
\begin{itemize}
    \item Pancreatic digest of gelatin 17.0 g
    \item Peptones (meat and casein) 3.0 g
    \item Lactose 10.0 g
    \item Bile salts (No. 3) 1.5 g
    \item Sodium chloride 5.0 g
    \item Neutral red 0.03 g
    \item Crystal violet 0.001 g
    \item Agar 13.5 g
    \item น้ำกลั่น ปรับปริมาตรเป็น 1,000 mL (pH $7.1 \pm 0.2$)
\end{itemize}

\section{การเตรียมสีย้อมชีวภาพมาตรฐาน}
\noindent\textbf{1. Crystal Violet Solution:} ละลาย Crystal violet 2.0 g ใน Ethyl alcohol 95\% ปริมาตร 20 mL ผสมกับสารละลาย Ammonium oxalate 0.8 g ในน้ำกลั่น 80 mL

\noindent\textbf{2. Gram's Iodine Solution:} บด Iodine crystals 1.0 g ร่วมกับ Potassium iodide (KI) 2.0 g ในโกร่งบดยา ค่อยๆ เติมน้ำกลั่นจนละลายหมดแล้วปรับปริมาตรเป็น 300 mL

\noindent\textbf{3. Safranin O Solution:} ละลาย Safranin O 2.5 g ใน Ethyl alcohol 95\% ปริมาตร 100 mL นำสารละลายหลัก 10 mL ผสมกับน้ำกลั่น 90 mL
